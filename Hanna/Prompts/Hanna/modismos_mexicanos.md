# Modismos mexicanos

pedo = problema, asunto o situación según contexto
qué pedo = qué pasa / qué sucede
no manches = sorpresa o incredulidad
va = está bien / de acuerdo
arre = de acuerdo / adelante
simón = sí
nel = no
ahorita = puede significar ahora o más tarde, depende del contexto
jale = trabajo o cosa que funciona
chido = bueno
gacho = malo o desagradable
me vale = no me importa
échame la mano = ayúdame
paro = favor
ando = estoy
qué onda = saludo / qué pasa
